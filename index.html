function convertDocxToPdf() { const file = document.getElementById('docxFile').files[0]; const msg = 'docxMsg'; if (!file) { showMsg(msg, 'Select a Word file (.docx)', true); return; } if (!file.name.toLowerCase().endsWith('.docx')) { showMsg(msg, 'Please select a valid .docx file', true); return; } if (file.size > 5242880) { showMsg(msg, 'File is ' + bytesToMB(file.size) + ' MB. Max is 5 MB.', true); return; } if (mammothLoadFailed || typeof mammoth === 'undefined') { showMsg( msg, 'The Word conversion library could not load. Check your internet connection and reload the page.', true ); return; } showMsg(msg, 'Converting... please wait', false); const reader = new FileReader(); reader.onload = function(e) { mammoth.convertToHtml({ arrayBuffer: e.target.result }) .then(function(result) { const element = document.createElement('div'); element.innerHTML = result.value; const opt = { margin: 10, filename: file.name.replace(/\.docx$/i, '') + '.pdf', image: { type: 'jpeg', quality: 0.98 }, html2canvas: { scale: 2 }, jsPDF: { orientation: 'portrait', unit: 'mm', format: 'a4' } }; return html2pdf() .set(opt) .from(element) .save(); }) .then(function() { showMsg(msg, 'PDF downloaded!', false); }) .catch(function(err) { console.error('DOCX conversion error:', err); showMsg(msg, 'Error converting file: ' + err.message, true); }); }; reader.onerror = function() { showMsg(msg, 'Could not read the selected file.', true); }; reader.readAsArrayBuffer(file); }
